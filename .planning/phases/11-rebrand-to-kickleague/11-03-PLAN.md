<task type="auto">
  <name>Task 2: Create OG image generator for social previews</name>
  <files>src/app/opengraph-image.tsx</files>
  <action>
    Create `src/app/opengraph-image.tsx` using Next.js ImageResponse API.

    Design requirements (from locked decisions):
    - Dark card with wordmark centered, tagline below
    - Matching site dark theme (#0a0a0f background)
    - Standard OG image size: 1200x630

    Implementation:
    ```typescript
    import { ImageResponse } from 'next/og';

    export const alt = 'KickLeague - Football intelligence, visualized';
    export const size = { width: 1200, height: 630 };
    export const contentType = 'image/png';

    export default async function Image() {
      return new ImageResponse(
        (
          <div
            style={{
              display: 'flex',
              flexDirection: 'column',
              alignItems: 'center',
              justifyContent: 'center',
              width: '100%',
              height: '100%',
              background: '#0a0a0f',
              color: 'white',
              fontFamily: 'sans-serif',
            }}
          >
            {/* Football icon + wordmark text */}
            <div style={{ display: 'flex', alignItems: 'center', gap: '16px' }}>
              {/* Inline football icon as SVG in JSX */}
              <svg
                width="64"
                height="64"
                viewBox="0 0 32 32"
                fill="none"
                style={{ flexShrink: 0 }}
              >
                {/* Minimal outline football -- circle + panel seams */}
                <circle cx="16" cy="16" r="14" stroke="white" strokeWidth="2" />
                <path d="M16 2 L16 30" stroke="white" strokeWidth="1.5" opacity="0.4" />
                <path d="M2 16 L30 16" stroke="white" strokeWidth="1.5" opacity="0.4" />
                <ellipse cx="16" cy="16" rx="14" ry="8" stroke="white" strokeWidth="1.5" opacity="0.4" />
              </svg>
              <div
                style={{
                  fontSize: 72,
                  fontWeight: 800,
                  fontStyle: 'italic',
                  letterSpacing: '-1px',
                }}
              >
                KickLeague
              </div>
            </div>
            <div
              style={{
                fontSize: 28,
                marginTop: 20,
                opacity: 0.6,
                fontWeight: 400,
              }}
            >
              Football intelligence, visualized
            </div>
          </div>
        ),
        { ...size }
      );
    }
    ```

    **IMPORTANT:** ImageResponse uses Satori which only supports flexbox. Do NOT use CSS grid, position: absolute, or other non-flex properties. Keep the layout simple with flex containers.

    The exact SVG paths for the football icon should match the icon-only SVG from Plan 02 but rendered inline as JSX (since ImageResponse cannot load external files without readFile).
  </action>
