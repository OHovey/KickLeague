    return {
      '@context': 'https://schema.org',
      '@type': 'ItemList',
      itemListElement: items.map((item) => ({
        '@type': 'ListItem',
        position: item.position,
        name: item.name,
        ...(item.url && { url: item.url.startsWith('http') ? item.url : `${SITE_URL}${item.url}` }),
      })),
    };
  }

**Add sitemap query** to `src/lib/seo/sitemap-queries.ts`:

```typescript
export async function getStatsSitemapEntries(): Promise<SitemapEntry[]> {
  // Stats pages are static combinations of leagues x stat types
  // No database query needed -- just enumerate all 15 combinations
  const statTypes = ['top-scorers', 'top-assists', 'disciplinary'];

  if (!isDatabaseConfigured()) return [];

  const leagueRows = await getDb()
    .select({ slug: leagues.slug })
    .from(leagues);

  const now = new Date();
  const entries: SitemapEntry[] = [];

  for (const league of leagueRows) {
    for (const stat of statTypes) {
      entries.push({
        path: `/leagues/${league.slug}/stats/${stat}`,
        routeKey: '/leagues/[slug]/stats/[stat]',
        params: { slug: league.slug, stat },
        lastmod: now,
      });
    }
  }

  return entries;
}
```
